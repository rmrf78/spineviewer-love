cmake_minimum_required(VERSION 3.22)
project(SpineLoveEX VERSION 1.0.0 LANGUAGES C CXX)
if(MSVC)
    file(TO_NATIVE_PATH "${CMAKE_SOURCE_DIR}" source_prefix)
    add_compile_options(/experimental:deterministic "/pathmap:${source_prefix}=spinelove")
endif()
if(WIN32)
    add_executable(SpineLoveEX WIN32 main/launcher.cpp main/app.rc)
    target_compile_features(SpineLoveEX PRIVATE cxx_std_17)
    target_compile_definitions(SpineLoveEX PRIVATE UNICODE _UNICODE)
    target_link_libraries(SpineLoveEX PRIVATE shell32)
    if(MSVC)
        target_compile_options(SpineLoveEX PRIVATE /utf-8)
        set_property(TARGET SpineLoveEX PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
include(cmake/SpineRuntimes.cmake)
add_library(spinelove_playback STATIC main/spine_runtime_registry.cpp main/runtime_shared/sl_skeleton_probe.cpp)
target_compile_features(spinelove_playback PUBLIC cxx_std_17)
target_include_directories(spinelove_playback PUBLIC "${PROJECT_SOURCE_DIR}/main")
target_link_libraries(spinelove_playback PUBLIC SpineLove::Runtime)
set_target_properties(spinelove_playback PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(spinelove_playback PRIVATE /utf-8)
endif()
add_library(SpineLove::Playback ALIAS spinelove_playback)
add_subdirectory(main/qt)
