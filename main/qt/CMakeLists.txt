set(SPINELOVE_QT_VERSION "6.8.3" CACHE STRING "Qt version")
find_package(Qt6 ${SPINELOVE_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 ShaderTools)
set(CMAKE_AUTOMOC ON)
include("${PROJECT_SOURCE_DIR}/cmake/Live2DWindows.cmake")
add_library(spinelove_qt_textures STATIC render/texture_loader.h render/texture_loader.cpp)
target_compile_features(spinelove_qt_textures PUBLIC cxx_std_17)
target_include_directories(spinelove_qt_textures PRIVATE "${PROJECT_SOURCE_DIR}/third_party/stb")
target_link_libraries(spinelove_qt_textures PUBLIC Qt6::Gui)
add_library(spinelove_qt_scene STATIC render/scene_snapshot.h render/scene_snapshot.cpp)
target_compile_features(spinelove_qt_scene PUBLIC cxx_std_17)
target_link_libraries(spinelove_qt_scene PUBLIC spinelove_qt_textures Qt6::Core Qt6::Gui)
add_library(spinelove_qt_controller STATIC
    core/viewer_controller.h core/viewer_controller.cpp core/viewer_export.cpp core/viewer_window.cpp
    core/asset_library.h core/asset_library.cpp core/export_service.h core/export_service.cpp
    render/slot_outline.h render/slot_outline.cpp)
target_compile_features(spinelove_qt_controller PUBLIC cxx_std_17)
target_include_directories(spinelove_qt_controller PRIVATE "${PROJECT_SOURCE_DIR}/third_party/stb")
target_link_libraries(spinelove_qt_controller PUBLIC SpineLove::Playback SpineLove::QtLive2D spinelove_qt_scene Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick)
set_target_properties(spinelove_qt_textures spinelove_qt_scene spinelove_qt_controller PROPERTIES POSITION_INDEPENDENT_CODE ON)
qt_add_executable(spinelove_qt main.cpp render/spine_scene.h render/spine_scene.cpp)
if(WIN32)
    set_target_properties(spinelove_qt PROPERTIES WIN32_EXECUTABLE TRUE)
    target_sources(spinelove_qt PRIVATE "${PROJECT_SOURCE_DIR}/main/app.rc")
    target_include_directories(spinelove_qt PRIVATE "${PROJECT_SOURCE_DIR}/main")
endif()
target_compile_features(spinelove_qt PRIVATE cxx_std_17)
target_include_directories(spinelove_qt PRIVATE "${PROJECT_SOURCE_DIR}/third_party/stb")
target_link_libraries(spinelove_qt PRIVATE spinelove_qt_controller Qt6::GuiPrivate Qt6::QuickControls2)
if(MSVC)
    foreach(target spinelove_qt spinelove_qt_controller spinelove_qt_textures spinelove_qt_scene)
        target_compile_options(${target} PRIVATE /utf-8)
    endforeach()
endif()
file(GLOB UI_QML CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ui/*.qml")
qt_add_resources(spinelove_qt "viewer_ui" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES Main.qml ${UI_QML})
qt_add_resources(spinelove_qt "viewer_icon" PREFIX "/" BASE "${PROJECT_SOURCE_DIR}" FILES "${PROJECT_SOURCE_DIR}/app.png" "${PROJECT_SOURCE_DIR}/main/app.ico")
qt_add_resources(spinelove_qt "viewer_languages" PREFIX "/lang" BASE "${CMAKE_CURRENT_SOURCE_DIR}/lang" FILES lang/zh_CN.txt)
qt_add_shaders(spinelove_qt "viewer_shaders" PREFIX "/shaders" BASE "${CMAKE_CURRENT_SOURCE_DIR}/shaders" FILES shaders/spine.vert shaders/spine.frag)
add_custom_command(TARGET spinelove_qt POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROJECT_SOURCE_DIR}/main/NotoSansSC-Regular.ttf" "$<TARGET_FILE_DIR:spinelove_qt>/NotoSansSC-Regular.ttf"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:spinelove_qt>/render_d3d11/shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROJECT_SOURCE_DIR}/main/render_d3d11/shaders/sprite.hlsl" "$<TARGET_FILE_DIR:spinelove_qt>/render_d3d11/shaders/sprite.hlsl")
